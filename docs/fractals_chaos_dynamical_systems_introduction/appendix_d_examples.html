実行と検査の方法

サンプルは examples フォルダーで実行する（はじめに を参照）。多くのサンプルは、共通モジュールや、ほかの章のサンプルの関数を import する。個別にダウンロードした場合は、一覧の「必要なファイル」の列に示したファイルも同じフォルダーに置く。まとめて使う場合は、ZIP ファイルを使うと簡単である。

サンプルは、flake8（PEP 8 などのコーディング規約の検査、1 行は 120 文字まで）と mypy（型ヒントの検査、strict モード）で違反がないことを確認している。examples フォルダーで次のコマンドを実行すると、setup.cfg の設定で検査できる。

1$ cd examples
2$ python -m pip install flake8 mypy
3$ python -m flake8 .
4$ python -m mypy .

matplotlib の 3 次元グラフの機能（mpl_toolkits）は型ヒントの情報を持たないため、setup.cfg でこの import の検査を省略している。mypy は Windows ではシステムの文字コードで設定ファイルを読むので、setup.cfg のコメントは英語で書いている。

ode.py

 1"""常微分方程式 dx/dt = f(x) を数値的に解く、共通モジュール（第 3 章）.
 2
 3* euler_step()   オイラー法で 1 ステップ進める
 4* rk4_step()     4 次のルンゲ＝クッタ法で 1 ステップ進める
 5* integrate()    指定した方法で n ステップ進め、途中の状態をすべて返す
 6
 7状態 x は NumPy の配列で、最後の軸が状態の成分を表す。たとえばローレンツ方程式の
 8状態は形が (3,) の配列である。形が (1000, 3) の配列を渡せば、1000 個の初期値を
 9まとめて（ベクトル化して）計算できる。
10"""
11
12from collections.abc import Callable
13
14import numpy as np
15from numpy.typing import NDArray
16
17Vector = NDArray[np.float64]
18VectorField = Callable[[Vector], Vector]
19Stepper = Callable[[VectorField, Vector, float], Vector]
20
21
22def euler_step(f: VectorField, x: Vector, dt: float) -> Vector:
23    """オイラー法: 現在の傾き f(x) のまま dt だけ進む."""
24    return x + dt * f(x)
25
26
27def rk4_step(f: VectorField, x: Vector, dt: float) -> Vector:
28    """4 次のルンゲ＝クッタ法: 区間内の 4 か所の傾きを重み付きで平均して進む."""
29    k1 = f(x)
30    k2 = f(x + 0.5 * dt * k1)
31    k3 = f(x + 0.5 * dt * k2)
32    k4 = f(x + dt * k3)
33    return x + dt / 6.0 * (k1 + 2.0 * k2 + 2.0 * k3 + k4)
34
35
36def integrate(f: VectorField, x0: Vector, dt: float, n: int, step: Stepper = rk4_step) -> Vector:
37    """初期値 x0 から n ステップ進め、x0 を含む n + 1 個の状態を返す.
38
39    戻り値の形は (n + 1,) + x0.shape である。
40    """
41    xs = np.empty((n + 1,) + x0.shape)
42    xs[0] = x0
43    x = x0.astype(np.float64)
44    for i in range(n):
45        x = step(f, x, dt)
46        xs[i + 1] = x
47    return xs

 19def logistic(r: float) -> Map:
 20    """パラメータ r のロジスティック写像 f(x) = r x (1 - x) を返す."""

 26def cobweb(ax: Axes, f: Map, x0: float, n: int, lo: float = 0.0, hi: float = 1.0) -> None:
 27    """写像 f のグラフ、対角線 y = x、x0 から n 回反復したクモの巣図を ax に描く.

 58def superstable_parameters(count: int) -> list[float]:
 59    """周期 2^n の超安定な軌道を持つパラメータ R_0, R_1, ... を count 個求める.
 60
 61    超安定な軌道は、f の微分が 0 になる点 x = 1/2 を通る。そこで
 62    f_r を 2^n 回適用した値が 1/2 に戻る r を、割線法で解く。初期値は、
 63    それまでに求めた R から、間隔がほぼ一定の比で縮むことを使って予想する。
 64    """
 65    def g(r: float, n: int) -> float:
 66        x = 0.5
 67        for _ in range(2**n):
 68            x = r * x * (1.0 - x)
 69        return x - 0.5
 70
 71    params = [2.0, 1.0 + np.sqrt(5.0)]
 72    while len(params) < count:
 73        n = len(params)
 74        guess = params[-1] + (params[-1] - params[-2]) / 4.7
 75        r0, r1 = guess, guess + 1e-6 * (params[-1] - params[-2])
 76        for _ in range(50):
 77            g0, g1 = g(r0, n), g(r1, n)
 78            if g1 == g0:
 79                break
 80            r0, r1 = r1, r1 - g1 * (r1 - r0) / (g1 - g0)
 81            if abs(r1 - r0) < 1e-15:
 82                break
 83        params.append(float(r1))
 84    return params[:count]

22def baker_image(image: NDArray[np.float64]) -> NDArray[np.float64]:
23    """画像にパイこね変換 B(x, y) = (2x mod 1, (y + floor(2x)) / 2) を 1 回施す.

26def clifford(x: Array, y: Array, params: Params) -> tuple[Array, Array]:
27    """クリフォード・アトラクタの写像."""
28    a, b, c, d = params
29    return np.sin(a * y) + c * np.cos(a * x), np.sin(b * x) + d * np.cos(b * y)

33    """ピーター・デ・ヨング・アトラクタの写像."""

35    return np.sin(a * y) - np.cos(b * x), np.sin(c * x) - np.cos(d * y)

45        x, y = f(x, y, params)

47    ys = np.empty((n_steps, n_orbits))
48    for i in range(n_steps):

 23def double_pendulum(state: Vector) -> Vector:
 24    """二重振り子の運動方程式（質量と長さがすべて 1 の場合）の右辺."""
 25    t1, t2, w1, w2 = state[..., 0], state[..., 1], state[..., 2], state[..., 3]
 26    d = t1 - t2
 27    den = 3.0 - np.cos(2.0 * d)
 28    a1 = (-3.0 * G * np.sin(t1) - G * np.sin(t1 - 2.0 * t2)
 29          - 2.0 * np.sin(d) * (w2**2 + w1**2 * np.cos(d))) / den
 30    a2 = 2.0 * np.sin(d) * (2.0 * w1**2 + 2.0 * G * np.cos(t1) + w2**2 * np.cos(d)) / den
 31    return np.stack([w1, w2, a1, a2], axis=-1)

 34def energy(state: Vector) -> Vector:
 35    """力学的エネルギー（運動エネルギーと位置エネルギーの和）."""
 36    t1, t2, w1, w2 = state[..., 0], state[..., 1], state[..., 2], state[..., 3]
 37    kinetic = w1**2 + 0.5 * w2**2 + w1 * w2 * np.cos(t1 - t2)
 38    potential = -G * (2.0 * np.cos(t1) + np.cos(t2))
 39    result: Vector = kinetic + potential
 40    return result

 22def box_count(points: Points, eps: float) -> int:
 23    """点の集まり points（形は (n, 2)、座標は [0, 1] の範囲）と重なる、一辺 eps のボックスの数を返す."""
 24    boxes = np.floor(points / eps).astype(np.int64)

 19def cantor_intervals(level: int) -> list[tuple[float, float]]:
 20    """カントール集合を作る途中の、level 段階目の区間の一覧を返す."""
 21    intervals = [(0.0, 1.0)]
 22    for _ in range(level):
 23        next_intervals = []
 24        for a, b in intervals:
 25            third = (b - a) / 3.0
 26            next_intervals += [(a, a + third), (b - third, b)]  # 真ん中の 1/3 を取り除く
 27        intervals = next_intervals

 31def koch_curve(level: int, start: complex = 0.0, end: complex = 1.0) -> Points:
 32    """コッホ曲線の level 段階目の折れ線の頂点を、複素数の配列で返す.
 33
 34    線分を 3 等分し、真ん中の部分を、正三角形の 2 辺で置き換える。
 35    """
 36    points = np.array([start, end])
 37    rotate = np.exp(1j * np.pi / 3.0)  # 60 度の回転
 38    for _ in range(level):
 39        a, b = points[:-1], points[1:]
 40        d = (b - a) / 3.0
 41        p1 = a + d
 42        peak = p1 + d * rotate
 43        p2 = a + 2.0 * d
 44        new = np.stack([a, p1, peak, p2], axis=1).ravel()
 45        points = np.append(new, points[-1])
 46    return points

 20class Affine:
 21    """平面のアフィン変換 x -> A x + t."""
 22
 23    def __init__(self, a: float, b: float, c: float, d: float, e: float, f: float) -> None:
 24        """行列 A = [[a, b], [c, d]] と平行移動 t = (e, f) を指定する."""

 55def deterministic_iteration(maps: list[Affine], points: Array, n: int) -> Array:

 62def chaos_game(maps: list[Affine], weights: list[float] | None, n: int, n_walkers: int = 1000,
 63               seed: int = 0) -> Array:
 64    """カオスゲーム: 毎回 1 つの写像を確率的に選んで適用し、訪れた点を記録する.
 65
 66    1 個の点で n 回反復する代わりに、n_walkers 個の点を同時に動かして速く計算する。
 67    最初の 20 回は、アトラクタに近づくまでの過渡状態として捨てる。
 68    """
 69    rng = np.random.default_rng(seed)
 70    p = np.asarray(weights) if weights is not None else np.full(len(maps), 1.0 / len(maps))
 71    x = rng.random((n_walkers, 2))
 72    steps = n // n_walkers
 73    result = np.empty((steps, n_walkers, 2))
 74    for i in range(steps + 20):
 75        choice = rng.choice(len(maps), size=n_walkers, p=p)
 76        new = np.empty_like(x)
 77        for k, f in enumerate(maps):
 78            mask = choice == k
 79            new[mask] = f(x[mask])
 80        x = new
 81        if i >= 20:
 82            result[i - 20] = x
 83    return result.reshape(-1, 2)

 88    dets = np.array([max(f.contraction(), 0.01) for f in maps])

 40def rewrite(axiom: str, rules: dict[str, str], iterations: int) -> str:

 48def turtle_segments(commands: str, angle: float, heading: float = 0.0) -> list[Segment]:
 49    """タートルグラフィックスの命令列を解釈し、引いた線分の一覧を返す."""
 50    x, y = 0.0, 0.0
 51    stack: list[tuple[float, float, float]] = []
 52    segments: list[Segment] = []
 53    for ch in commands:
 54        if ch in "FG":
 55            nx = x + math.cos(math.radians(heading))
 56            ny = y + math.sin(math.radians(heading))
 57            segments.append(((x, y), (nx, ny)))
 58            x, y = nx, ny
 59        elif ch == "+":
 60            heading += angle
 61        elif ch == "-":
 62            heading -= angle
 63        elif ch == "[":
 64            stack.append((x, y, heading))
 65        elif ch == "]":
 66            x, y, heading = stack.pop()
 67    return segments

 53    count = np.full(z.shape, float(max_iter))
 54    last = np.zeros_like(z)
 55    active = np.arange(z.size)
 56    for n in range(max_iter):
 57        z_act = step(z[active], c[active])
 58        escaped = np.abs(z_act) > radius
 59        idx = active[escaped]
 60        count[idx] = n + 1
 61        last[idx] = z_act[escaped]
 62        z[active] = z_act
 63        active = active[~escaped]
 64        if active.size == 0:
 65            break
 66    return count.reshape(z0.shape), last.reshape(z0.shape)

 69def smooth_count(count: Real, last: Complex, max_iter: int, degree: int = 2) -> Real:
 70    """連続的な脱出時間 ν = n + 1 - log(log|z|) / log(d) を返す（集合内の点は NaN）.
 71
 72    整数の脱出時間で色を付けると、色の境目が等高線のような帯（バンド）になる。
 73    脱出した直後の |z| の大きさを使って、帯の間を滑らかにつなぐ。d は写像の次数
 74    （z^2 + c なら 2）である。
 75    """
 76    with np.errstate(divide="ignore", invalid="ignore"):
 77        nu: Real = count + 1.0 - np.log(np.log(np.abs(last))) / np.log(degree)
 78    nu[count >= max_iter] = np.nan
 79    return nu

23def newton_basins(p: Poly, dp: Poly, roots: list[complex], z0: Complex, max_iter: int = 60,
24                  tol: float = 1e-6) -> tuple[NDArray[np.int64], NDArray[np.int64]]:
25    """各点が収束した解の番号（収束しなければ -1）と、収束までの反復回数を返す."""

 24def correlation_integral(points: Array, radii: Array, chunk: int = 500) -> Array:
 25    """各半径 r について、距離が r 未満の点の組の割合 C(r) を返す.
 26
 27    すべての組の距離を一度に計算するとメモリーを多く使うので、chunk 個ずつに分けて計算する。
 28    """
 29    n = len(points)
 30    counts = np.zeros(len(radii))
 31    for start in range(0, n, chunk):
 32        block = points[start:start + chunk]
 33        d = np.sqrt(((block[:, None, :] - points[None, :, :]) ** 2).sum(axis=-1))
 34        # 自分自身との組（距離 0）と、同じ組を 2 回数えるのを避けるため、番号が大きい相手だけを数える
 35        idx = np.arange(start, start + len(block))
 36        mask = idx[:, None] < np.arange(n)[None, :]
 37        dist = np.sort(d[mask])
 38        counts += np.searchsorted(dist, radii)
 39    result: Array = counts / (n * (n - 1) / 2)
 40    return result

 32def pendulum(state: Vector) -> Vector:
 33    """状態 (x, y, vx, vy) の時間微分を返す."""
 34    x, y, vx, vy = state[..., 0], state[..., 1], state[..., 2], state[..., 3]
 35    ax = -K * x - GAMMA * vx
 36    ay = -K * y - GAMMA * vy
 37    for mx, my in MAGNETS:
 38        dx, dy = mx - x, my - y
 39        r2 = dx * dx + dy * dy + H * H
 40        inv_r3 = 1.0 / (r2 * np.sqrt(r2))  # 1 / r^3（** 1.5 より速い）
 41        ax = ax + dx * inv_r3
 42        ay = ay + dy * inv_r3
 43    return np.stack([vx, vy, ax, ay], axis=-1)

 46def basins(x: NDArray[np.float64], y: NDArray[np.float64], dt: float = 0.05,

 49
 50    25 ステップごとに、磁石の近くで遅くなった点を「止まった」と判定し、計算の対象から
 51    外して速くする（そこまで遅くなると、摩擦のため、もう別の磁石には移れない）。最後まで止まらなかった点の番号は -1 にする。

 21def midpoint_displacement_1d(levels: int, hurst: float, rng: np.random.Generator) -> Array:
 22    """長さ 2^levels + 1 の稜線（高さの列）を作る."""
 23    n = 2**levels
 24    h = np.zeros(n + 1)
 25    scale = 1.0
 26    step = n
 27    while step > 1:
 28        half = step // 2
 29        mid = (h[:-1:step] + h[step::step]) / 2.0  # 各区間の両端の平均
 30        h[half::step] = mid + scale * rng.normal(size=mid.shape)
 31        scale *= 2.0 ** (-hurst)
 32        step = half
 33    return h

 36def diamond_square(levels: int, hurst: float, rng: np.random.Generator) -> Array:
 37    """ダイヤモンド・スクエア法（中点変位法の 2 次元版）で、(2^levels + 1) 四方の高さの地図を作る.
 38
 39    1. ダイヤモンド段階: 正方形の中心に、4 隅の平均にランダムな量を加えた高さを置く
 40    2. スクエア段階: 辺の中点に、周りの 4 点（端では 3 点）の平均にランダムな量を加えた高さを置く
 41    """
 42    n = 2**levels
 43    h = np.zeros((n + 1, n + 1))
 44    scale = 1.0
 45    step = n
 46    while step > 1:
 47        half = step // 2
 48        corners = (h[:-1:step, :-1:step] + h[:-1:step, step::step]
 49                   + h[step::step, :-1:step] + h[step::step, step::step])
 50        h[half::step, half::step] = corners / 4.0 + scale * rng.normal(size=corners.shape)
 51        for row_start, col_start in ((0, half), (half, 0)):
 52            rows = np.arange(row_start, n + 1, step)
 53            cols = np.arange(col_start, n + 1, step)
 54            r, c = np.meshgrid(rows, cols, indexing="ij")
 55            total = np.zeros(r.shape)
 56            count = np.zeros(r.shape)
 57            for dr, dc in ((-half, 0), (half, 0), (0, -half), (0, half)):
 58                rr, cc = r + dr, c + dc
 59                ok = (rr >= 0) & (rr <= n) & (cc >= 0) & (cc <= n)
 60                total[ok] += h[rr[ok], cc[ok]]
 61                count[ok] += 1
 62            h[r, c] = total / count + scale * rng.normal(size=r.shape)
 63        scale *= 2.0 ** (-hurst)
 64        step = half
 65    return h

 89            xe, ye = quadratic_map(coef, xe, ye)
 90            alive &= np.isfinite(x) & (np.abs(x) < 1e6) & (np.abs(y) < 1e6)
 91            d = np.hypot(xe - x, ye - y)
 92            d = np.where(alive & (d > 0), d, 1e-6)
 93            if n >= transient:
 94                lyap += np.log(d / 1e-6)
 95                xmin, xmax = np.minimum(xmin, x), np.maximum(xmax, x)
 96            # 2 つの点の距離を 1e-6 に戻す
 97            xe = x + (xe - x) * 1e-6 / d
 98            ye = y + (ye - y) * 1e-6 / d
 99            x = np.where(alive, x, 0.0)
100            y = np.where(alive, y, 0.0)
101    lyap /= n_steps - transient
102    # 発散せず、リアプノフ指数が正で、1 点に縮まらない（ある程度の広がりを持つ）ものを残す
103    good = np.nonzero(alive & (lyap > 0.05) & (xmax - xmin > 0.1))[0]
104    return [(coef[:, i], float(lyap[i])) for i in good[:n_found]]

 35def horseshoe(x: Array, y: Array) -> tuple[Array, Array]:
 36    """極座標の角度を 2 倍にし、距離はそのままにする（馬蹄形に折り曲げる）."""
 37    r = np.sqrt(x * x + y * y) + 1e-9

 41def swirl(x: Array, y: Array) -> tuple[Array, Array]:
 42    """原点からの距離に応じて回転させる（渦巻き）."""
 43    r2 = x * x + y * y
 44    s, c = np.sin(r2), np.cos(r2)
 45    return x * s - y * c, x * c + y * s

 50# アフィン変換の係数 (a, b, c, d, e, f): x' = a x + b y + e, y' = c x + d y + f
 51AFFINES = np.array([
 52    [0.56, -0.38, 0.38, 0.56, 0.20, 0.10],
 53    [0.45, 0.30, -0.30, 0.45, -0.55, 0.25],
 54    [-0.40, 0.25, 0.20, 0.50, 0.15, -0.60],

 59def flame(variation: Variation, n_points: int = 1_500_000, n_walkers: int = 5000, size: int = 400,
 60          span: float = 4.0, seed: int = 1) -> tuple[Array, Array]:
 61    """画素ごとの点の数と、色の番号の平均を返す（span は描く範囲の一辺の長さ）."""
 62    rng = np.random.default_rng(seed)
 63    x = rng.uniform(-1.0, 1.0, n_walkers)
 64    y = rng.uniform(-1.0, 1.0, n_walkers)
 65    c = rng.random(n_walkers)
 66    hist = np.zeros(size * size)
 67    color_sum = np.zeros(size * size)
 68    for n in range(n_points // n_walkers + 20):
 69        k = rng.integers(0, len(AFFINES), n_walkers)
 70        a, b, cc, d, e, f = AFFINES[k].T
 71        x, y = variation(a * x + b * y + e, cc * x + d * y + f)
 72        c = (c + COLORS[k]) / 2.0
 73        if n < 20:
 74            continue  # 最初の 20 回は、アトラクタに近づくまでの過渡状態として捨てる
 75        col = np.floor((x / span + 0.5) * size).astype(np.int64)
 76        row = np.floor((0.5 - y / span) * size).astype(np.int64)
 77        ok = (col >= 0) & (col < size) & (row >= 0) & (row < size)
 78        idx = row[ok] * size + col[ok]
 79        hist += np.bincount(idx, minlength=size * size)
 80        color_sum += np.bincount(idx, weights=c[ok], minlength=size * size)
 81    with np.errstate(invalid="ignore"):
 82        mean_color = np.where(hist > 0, color_sum / hist, 0.0)
 83    return hist.reshape(size, size), mean_color.reshape(size, size)

 86def render(hist: Array, mean_color: Array, cmap: str = "plasma", gamma: float = 2.5) -> Array:
 87    """点の数の対数で明るさを、色の番号の平均で色合いを決めた RGB 画像を返す（背景は黒）."""
 88    alpha = (np.log1p(hist) / np.log1p(hist.max())) ** (1.0 / gamma)
 89    rgb = plt.get_cmap(cmap)(mean_color)[..., :3]
 90    result: Array = rgb * alpha[..., None]
 91    return result
