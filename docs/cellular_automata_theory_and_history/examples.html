elementary_ca.py

ルール番号を指定して、基本セル・オートマトンの時空図を表示する。既定では中央の 1 セルだけが 1 の初期配置を使い、--random を指定するとランダムな初期配置を使う。

1python elementary_ca.py 30
2python elementary_ca.py 110 --width 301 --steps 150 --random
3python elementary_ca.py 90 --output rule90.png

 1"""基本セル・オートマトン (ECA) の時空図を描くサンプル.
 2
 3ルール番号 (0〜255) を指定すると, その規則で時間発展させた結果を
 4時空図 (横軸がセルの位置, 縦軸が時刻で, 下に行くほど新しい) として表示する.
 5
 6使い方の例::
 7
 8    python elementary_ca.py 30
 9    python elementary_ca.py 110 --width 301 --steps 150 --random
10    python elementary_ca.py 90 --output rule90.png
11"""
12
13from __future__ import annotations
14
15import argparse
16
17import numpy as np
18import numpy.typing as npt
19
20Cells = npt.NDArray[np.uint8]
21
22
23def rule_table(rule: int) -> Cells:
24    """ルール番号から局所規則の表を作る.
25
26    戻り値の k 番目の要素は, 近傍 (左, 中央, 右) を 2 進数 4 * 左 + 2 * 中央 + 右 と
27    みなしたときの値が k である場合の次の状態を表す.
28    """
29    if not 0 <= rule <= 255:
30        raise ValueError(f"ルール番号は 0〜255 で指定する: {rule}")
31    return np.array([(rule >> k) & 1 for k in range(8)], dtype=np.uint8)
32
33
34def step(cells: Cells, table: Cells) -> Cells:
35    """周期境界条件のもとで 1 ステップだけ時間発展させる."""
36    left = np.roll(cells, 1)    # 各セルの左隣の状態
37    right = np.roll(cells, -1)  # 各セルの右隣の状態
38    index = 4 * left + 2 * cells + right
39    result: Cells = table[index]
40    return result
41
42
43def evolve(rule: int, initial: Cells, steps: int) -> Cells:
44    """初期配置から steps ステップ分を計算し, 時空図を 2 次元配列で返す.
45
46    戻り値の形は (steps + 1, セル数) で, 0 行目が初期配置である.
47    """
48    table = rule_table(rule)
49    history = np.zeros((steps + 1, initial.size), dtype=np.uint8)
50    history[0] = initial
51    for t in range(steps):
52        history[t + 1] = step(history[t], table)
53    return history
54
55
56def single_seed(width: int) -> Cells:
57    """中央の 1 セルだけが 1 で, 残りが 0 の初期配置を作る."""
58    cells = np.zeros(width, dtype=np.uint8)
59    cells[width // 2] = 1
60    return cells
61
62
63def random_cells(width: int, density: float = 0.5, seed: int | None = None) -> Cells:
64    """各セルが確率 density で 1 になるランダムな初期配置を作る."""
65    rng = np.random.default_rng(seed)
66    return (rng.random(width) < density).astype(np.uint8)
67
68
69def main() -> None:
70    parser = argparse.ArgumentParser(description="基本セル・オートマトンの時空図を描く")
71    parser.add_argument("rule", type=int, help="ルール番号 (0〜255)")
72    parser.add_argument("--width", type=int, default=201, help="セルの数")
73    parser.add_argument("--steps", type=int, default=100, help="計算するステップ数")
74    parser.add_argument("--random", action="store_true", help="ランダムな初期配置を使う")
75    parser.add_argument("--seed", type=int, default=None, help="乱数のシード")
76    parser.add_argument("--output", default=None, help="画像の保存先 (省略時は画面に表示)")
77    args = parser.parse_args()
78
79    if args.random:
80        initial = random_cells(args.width, seed=args.seed)
81    else:
82        initial = single_seed(args.width)
83    history = evolve(args.rule, initial, args.steps)
84
85    import matplotlib.pyplot as plt
86
87    fig, ax = plt.subplots(figsize=(8, 8 * history.shape[0] / history.shape[1]))
88    ax.imshow(history, cmap="binary", interpolation="nearest")
89    ax.set_title(f"Rule {args.rule}")
90    ax.set_xlabel("position")
91    ax.set_ylabel("time")
92    if args.output:
93        fig.savefig(args.output, dpi=150, bbox_inches="tight")
94    else:
95        plt.show()
96
97
98if __name__ == "__main__":
99    main()

game_of_life.py

ライフゲームをアニメーション表示する。--pattern で初期パターン（block、beehive、blinker、toad、glider、lwss、r_pentomino、gosper_gun）を、--rule で B/S 表記の規則を指定できる。

1python game_of_life.py --pattern glider
2python game_of_life.py --pattern gosper_gun --size 60 80
3python game_of_life.py --random 0.3 --rule B36/S23

  1"""ライフゲーム (および Life-like セル・オートマトン) のサンプル.
  2
  3B/S 表記で規則を指定できる. 既定値の B3/S23 がコンウェイのライフゲームである.
  4境界条件は周期境界 (上端と下端, 左端と右端がつながっている) とする.
  5
  6使い方の例::
  7
  8    python game_of_life.py --pattern glider
  9    python game_of_life.py --pattern gosper_gun --size 60 80
 10    python game_of_life.py --random 0.3 --rule B36/S23
 11"""
 12
 13from __future__ import annotations
 14
 15import argparse
 16
 17import numpy as np
 18import numpy.typing as npt
 19
 20Grid = npt.NDArray[np.uint8]
 21
 22# パターンは文字列で表す. "O" が生きているセル, "." が死んでいるセルである.
 23PATTERNS: dict[str, list[str]] = {
 24    "block": [
 25        "OO",
 26        "OO",
 27    ],
 28    "beehive": [
 29        ".OO.",
 30        "O..O",
 31        ".OO.",
 32    ],
 33    "blinker": [
 34        "OOO",
 35    ],
 36    "toad": [
 37        ".OOO",
 38        "OOO.",
 39    ],
 40    "glider": [
 41        ".O.",
 42        "..O",
 43        "OOO",
 44    ],
 45    "lwss": [
 46        ".O..O",
 47        "O....",
 48        "O...O",
 49        "OOOO.",
 50    ],
 51    "r_pentomino": [
 52        ".OO",
 53        "OO.",
 54        ".O.",
 55    ],
 56    "gosper_gun": [
 57        "........................O...........",
 58        "......................O.O...........",
 59        "............OO......OO............OO",
 60        "...........O...O....OO............OO",
 61        "OO........O.....O...OO..............",
 62        "OO........O...O.OO....O.O...........",
 63        "..........O.....O.......O...........",
 64        "...........O...O....................",
 65        "............OO......................",
 66    ],
 67}
 68
 69
 70def parse_rule(rule: str) -> tuple[frozenset[int], frozenset[int]]:
 71    """"B3/S23" のような B/S 表記を (誕生の条件, 生存の条件) に変換する."""
 72    birth_part, survive_part = rule.upper().split("/")
 73    if not (birth_part.startswith("B") and survive_part.startswith("S")):
 74        raise ValueError(f"B/S 表記ではない: {rule}")
 75    birth = frozenset(int(ch) for ch in birth_part[1:])
 76    survive = frozenset(int(ch) for ch in survive_part[1:])
 77    return birth, survive
 78
 79
 80def pattern_to_array(pattern: list[str]) -> Grid:
 81    """文字列で表したパターンを 0/1 の配列に変換する."""
 82    return np.array([[1 if ch == "O" else 0 for ch in row] for row in pattern], dtype=np.uint8)
 83
 84
 85def place(pattern: list[str], shape: tuple[int, int], top: int | None = None, left: int | None = None) -> Grid:
 86    """指定した大きさの盤面にパターンを置く. 位置を省略すると中央に置く."""
 87    grid = np.zeros(shape, dtype=np.uint8)
 88    cells = pattern_to_array(pattern)
 89    h, w = cells.shape
 90    if top is None:
 91        top = (shape[0] - h) // 2
 92    if left is None:
 93        left = (shape[1] - w) // 2
 94    grid[top:top + h, left:left + w] = cells
 95    return grid
 96
 97
 98def count_neighbors(grid: Grid) -> Grid:
 99    """各セルの周囲 8 セル (ムーア近傍から中心を除いたもの) にある生きたセルの数を数える."""
100    total = np.zeros_like(grid)
101    for dy in (-1, 0, 1):
102        for dx in (-1, 0, 1):
103            if dy == 0 and dx == 0:
104                continue
105            total += np.roll(grid, (dy, dx), axis=(0, 1))
106    return total
107
108
109def step(grid: Grid,
110         birth: frozenset[int] = frozenset({3}),
111         survive: frozenset[int] = frozenset({2, 3})) -> Grid:
112    """1 世代だけ時間発展させる."""
113    n = count_neighbors(grid)
114    born = (grid == 0) & np.isin(n, list(birth))
115    stays = (grid == 1) & np.isin(n, list(survive))
116    result: Grid = (born | stays).astype(np.uint8)
117    return result
118
119
120def main() -> None:
121    parser = argparse.ArgumentParser(description="ライフゲームをアニメーション表示する")
122    parser.add_argument("--rule", default="B3/S23", help="B/S 表記の規則 (既定値は B3/S23)")
123    parser.add_argument("--pattern", choices=sorted(PATTERNS), default="glider", help="初期パターン")
124    parser.add_argument("--random", type=float, default=None, metavar="DENSITY",
125                        help="指定すると, この密度のランダムな初期配置を使う")
126    parser.add_argument("--size", type=int, nargs=2, default=(40, 40), metavar=("ROWS", "COLS"), help="盤面の大きさ")
127    parser.add_argument("--steps", type=int, default=200, help="表示する世代数")
128    parser.add_argument("--interval", type=int, default=100, help="1 世代あたりの表示時間 [ms]")
129    args = parser.parse_args()
130
131    birth, survive = parse_rule(args.rule)
132    shape = (args.size[0], args.size[1])
133    if args.random is not None:
134        rng = np.random.default_rng()
135        grid = (rng.random(shape) < args.random).astype(np.uint8)
136    else:
137        grid = place(PATTERNS[args.pattern], shape)
138
139    import matplotlib.pyplot as plt
140    from matplotlib.animation import FuncAnimation
141    from matplotlib.artist import Artist
142
143    fig, ax = plt.subplots()
144    image = ax.imshow(grid, cmap="binary", vmin=0, vmax=1, interpolation="nearest")
145    ax.set_xticks([])
146    ax.set_yticks([])
147
148    state = {"grid": grid}
149
150    def update(frame: int) -> list[Artist]:
151        state["grid"] = step(state["grid"], birth, survive)
152        image.set_data(state["grid"])
153        ax.set_title(f"{args.rule}  generation {frame + 1}")
154        return [image]
155
156    # アニメーションのオブジェクトは表示が終わるまで変数に保持しておく必要がある.
157    animation = FuncAnimation(fig, update, frames=args.steps, interval=args.interval, repeat=False)
158    plt.show()
159    del animation
160
161
162if __name__ == "__main__":
163    main()

langtons_ant.py

ラングトンのアリを指定したステップ数だけ動かし、結果を表示する。既定の 11000 ステップでは、ハイウェイが伸び始めた様子が見られる。

1python langtons_ant.py
2python langtons_ant.py --steps 500 --size 40

 1"""ラングトンのアリのサンプル.
 2
 3白いセルの上ではアリが右に 90 度向きを変え, 黒いセルの上では左に 90 度向きを変える.
 4その後, いま居るセルの色を反転させてから, 向いている方向へ 1 セル進む.
 5何もない盤面から始めると, 約 1 万ステップの不規則な動きの後に「ハイウェイ」と
 6呼ばれる周期的な構造を作りながら一方向へ進み続ける.
 7
 8使い方の例::
 9
10    python langtons_ant.py
11    python langtons_ant.py --steps 500 --size 40
12"""
13
14from __future__ import annotations
15
16import argparse
17from dataclasses import dataclass
18
19import numpy as np
20import numpy.typing as npt
21
22Grid = npt.NDArray[np.uint8]
23
24# 向きは 0: 上, 1: 右, 2: 下, 3: 左 とする. 配列の行番号は下に行くほど大きい.
25DIRECTIONS: tuple[tuple[int, int], ...] = ((-1, 0), (0, 1), (1, 0), (0, -1))
26
27
28@dataclass
29class Ant:
30    """アリの位置 (行, 列) と向き."""
31
32    row: int
33    col: int
34    direction: int = 0
35
36
37def step(grid: Grid, ant: Ant) -> None:
38    """アリを 1 ステップ動かす. grid と ant をその場で書き換える."""
39    if grid[ant.row, ant.col] == 0:   # 白いセル: 右に曲がる
40        ant.direction = (ant.direction + 1) % 4
41    else:                             # 黒いセル: 左に曲がる
42        ant.direction = (ant.direction - 1) % 4
43    grid[ant.row, ant.col] ^= 1       # セルの色を反転する
44    d_row, d_col = DIRECTIONS[ant.direction]
45    rows, cols = grid.shape
46    ant.row = (ant.row + d_row) % rows  # 周期境界条件
47    ant.col = (ant.col + d_col) % cols
48
49
50def run(size: int, steps: int) -> tuple[Grid, Ant]:
51    """size × size の白い盤面の中央からアリを steps ステップ動かす."""
52    grid = np.zeros((size, size), dtype=np.uint8)
53    ant = Ant(size // 2, size // 2)
54    for _ in range(steps):
55        step(grid, ant)
56    return grid, ant
57
58
59def main() -> None:
60    parser = argparse.ArgumentParser(description="ラングトンのアリを動かして結果を表示する")
61    parser.add_argument("--size", type=int, default=160, help="盤面の一辺のセル数")
62    parser.add_argument("--steps", type=int, default=11000, help="ステップ数")
63    parser.add_argument("--output", default=None, help="画像の保存先 (省略時は画面に表示)")
64    args = parser.parse_args()
65
66    grid, ant = run(args.size, args.steps)
67
68    import matplotlib.pyplot as plt
69
70    fig, ax = plt.subplots(figsize=(6, 6))
71    ax.imshow(grid, cmap="binary", interpolation="nearest")
72    ax.plot(ant.col, ant.row, "r.", markersize=8)
73    ax.set_title(f"Langton's ant after {args.steps} steps")
74    ax.set_xticks([])
75    ax.set_yticks([])
76    if args.output:
77        fig.savefig(args.output, dpi=150, bbox_inches="tight")
78    else:
79        plt.show()
80
81
82if __name__ == "__main__":
83    main()

traffic_nasch.py

ナーゲル・シュレッケンベルク・モデルの時空図を表示する。--diagram を指定すると、密度を変えながら流量を求めて基本図を描く。

1python traffic_nasch.py
2python traffic_nasch.py --density 0.1 --p 0.3
3python traffic_nasch.py --diagram

  1"""ナーゲル・シュレッケンベルク・モデルによる交通流シミュレーション.
  2
  3周回道路 (周期境界条件) を L 個のセルに区切り, 各セルに車が高々 1 台入る.
  4各車の速度は 0〜v_max の整数で, 毎ステップ次の 4 つの規則を全車に同時に適用する.
  5
  61. 加速: v ← min(v + 1, v_max)
  72. 減速: v ← min(v, 前の車までの空きセル数)
  83. ランダムな減速: 確率 p で v ← max(v - 1, 0)
  94. 移動: 位置 ← 位置 + v
 10
 11使い方の例::
 12
 13    python traffic_nasch.py                       # 時空図を表示する
 14    python traffic_nasch.py --density 0.1 --p 0.3
 15    python traffic_nasch.py --diagram             # 基本図 (密度と流量の関係) を表示する
 16"""
 17
 18from __future__ import annotations
 19
 20import argparse
 21from dataclasses import dataclass
 22
 23import numpy as np
 24import numpy.typing as npt
 25
 26IntArray = npt.NDArray[np.int64]
 27
 28EMPTY = -1  # 時空図で車が居ないセルを表す値
 29
 30
 31@dataclass
 32class Road:
 33    """周回道路の状態. positions と velocities の i 番目が i 台目の車を表す."""
 34
 35    length: int
 36    positions: IntArray
 37    velocities: IntArray
 38
 39
 40def make_road(length: int, n_cars: int, rng: np.random.Generator) -> Road:
 41    """車をランダムな位置に速度 0 で配置した道路を作る."""
 42    positions = np.sort(rng.choice(length, size=n_cars, replace=False)).astype(np.int64)
 43    velocities = np.zeros(n_cars, dtype=np.int64)
 44    return Road(length, positions, velocities)
 45
 46
 47def step(road: Road, v_max: int, p: float, rng: np.random.Generator) -> None:
 48    """全車に 4 つの規則を同時に適用する. road をその場で書き換える.
 49
 50    車は追い越しをしないので, 配列上の並び順は周回方向の並び順のまま保たれる.
 51    """
 52    pos = road.positions
 53    vel = road.velocities
 54    # 前の車 (配列上で次の要素, 最後の車にとっては先頭の車) までの空きセル数
 55    gaps = (np.roll(pos, -1) - pos - 1) % road.length
 56    if pos.size == 1:
 57        gaps[:] = road.length - 1
 58    vel = np.minimum(vel + 1, v_max)              # 1. 加速
 59    vel = np.minimum(vel, gaps)                   # 2. 減速
 60    slow = rng.random(vel.size) < p
 61    vel = np.where(slow, np.maximum(vel - 1, 0), vel)  # 3. ランダムな減速
 62    road.positions = (pos + vel) % road.length    # 4. 移動
 63    road.velocities = vel
 64
 65
 66def snapshot(road: Road) -> IntArray:
 67    """道路の各セルについて, 車が居れば速度を, 居なければ EMPTY を入れた配列を返す."""
 68    cells = np.full(road.length, EMPTY, dtype=np.int64)
 69    cells[road.positions] = road.velocities
 70    return cells
 71
 72
 73def simulate(length: int, density: float, v_max: int, p: float, steps: int, seed: int | None = None) -> IntArray:
 74    """時空図 (形は (steps + 1, length)) を計算する."""
 75    rng = np.random.default_rng(seed)
 76    road = make_road(length, max(1, round(density * length)), rng)
 77    history = np.empty((steps + 1, length), dtype=np.int64)
 78    history[0] = snapshot(road)
 79    for t in range(steps):
 80        step(road, v_max, p, rng)
 81        history[t + 1] = snapshot(road)
 82    return history
 83
 84
 85def mean_flow(length: int, density: float, v_max: int, p: float,
 86              transient: int = 500, measure: int = 500, seed: int | None = None) -> float:
 87    """十分に時間が経った後の流量 (1 ステップに 1 セルあたりを通過する車の台数) を求める.
 88
 89    流量は 密度 × 平均速度 = (全車の速度の和) / 道路のセル数 で計算できる.
 90    """
 91    rng = np.random.default_rng(seed)
 92    n_cars = round(density * length)
 93    if n_cars == 0:
 94        return 0.0
 95    road = make_road(length, n_cars, rng)
 96    for _ in range(transient):

102    return total / measure

 1"""HPP モデル (格子ガス・オートマトン) の簡易実装.
 2
 3正方格子の各ノードには, 東・北・西・南の 4 方向へ動く粒子が方向ごとに高々 1 個ずつ存在できる.
 41 ステップは次の 2 段階からなる.
 5
 61. 衝突: 正面衝突する 2 粒子 (東と西のみ, または北と南のみ) を 90 度回転させる.
 72. 移動: 各粒子を向いている方向の隣のノードへ移す.
 8
 9衝突の前後で粒子数と運動量が保存される. 境界条件は周期境界とする.
10
11使い方の例::
12
13    python lattice_gas_hpp.py
14    python lattice_gas_hpp.py --size 256 --steps 60
15"""
16
17from __future__ import annotations
18
19import argparse
20
21import numpy as np
22import numpy.typing as npt
23
24BoolGrid = npt.NDArray[np.bool_]
25FloatGrid = npt.NDArray[np.float64]
26
27# 4 方向の粒子の有無を形 (4, 行数, 列数) の真偽値配列で表す. 配列の行番号は下に行くほど大きい.
28EAST, NORTH, WEST, SOUTH = 0, 1, 2, 3
29
30
31def collide(state: BoolGrid) -> BoolGrid:
32    """正面衝突する粒子の組を 90 度回転させた状態を返す."""
33    e, n, w, s = state
34    horizontal = e & w & ~n & ~s  # 東西からの正面衝突
35    vertical = n & s & ~e & ~w    # 南北からの正面衝突
36    flip = horizontal | vertical  # 該当するノードでは 4 方向すべての有無を反転すればよい
37    result: BoolGrid = state ^ flip
38    return result
39
40
41def propagate(state: BoolGrid) -> BoolGrid:
42    """各粒子を隣のノードへ移動させた状態を返す."""
43    moved = np.empty_like(state)
44    moved[EAST] = np.roll(state[EAST], 1, axis=1)
45    moved[WEST] = np.roll(state[WEST], -1, axis=1)
46    moved[NORTH] = np.roll(state[NORTH], -1, axis=0)
47    moved[SOUTH] = np.roll(state[SOUTH], 1, axis=0)
48    return moved
49
50
51def step(state: BoolGrid) -> BoolGrid:
52    """衝突と移動をこの順に行う."""
53    return propagate(collide(state))
54
55
56def initial_state(size: int, background: float, bump: float, radius: int, seed: int | None = None) -> BoolGrid:
57    """一様な背景の中央に, 粒子の密度が高い円形の領域を置いた初期状態を作る."""
58    rng = np.random.default_rng(seed)
59    y, x = np.mgrid[0:size, 0:size]
60    inside = (y - size / 2) ** 2 + (x - size / 2) ** 2 < radius ** 2
61    probability = np.where(inside, bump, background)
62    state: BoolGrid = rng.random((4, size, size)) < probability
63    return state
64
65
66def coarse_density(state: BoolGrid, block: int) -> FloatGrid:
67    """block × block のノードごとに平均した粒子密度を返す. 微視的なゆらぎをならして見るために使う."""
68    counts = state.sum(axis=0).astype(np.float64)
69    rows, cols = counts.shape
70    reshaped = counts[:rows - rows % block, :cols - cols % block].reshape(rows // block, block, cols // block, block)
71    result: FloatGrid = reshaped.mean(axis=(1, 3))
72    return result
73

75def main() -> None:
76    parser = argparse.ArgumentParser(description="HPP モデルで密度の波が広がる様子を表示する")
77    parser.add_argument("--size", type=int, default=256, help="格子の一辺のノード数")
78    parser.add_argument("--steps", type=int, default=60, help="ステップ数")
79    parser.add_argument("--block", type=int, default=4, help="密度を平均するブロックの一辺")
80    args = parser.parse_args()
81
82    state = initial_state(args.size, background=0.2, bump=0.7, radius=args.size // 16, seed=1)
83    for _ in range(args.steps):
84        state = step(state)
85
86    import matplotlib.pyplot as plt
87
88    fig, ax = plt.subplots(figsize=(6, 6))
89    ax.imshow(coarse_density(state, args.block), cmap="viridis", interpolation="nearest")
90    ax.set_title(f"HPP lattice gas, step {args.steps}")
91    ax.set_xticks([])
92    ax.set_yticks([])
93    plt.show()
