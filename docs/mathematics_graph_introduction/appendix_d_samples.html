 3同じ無向グラフを辺リスト・隣接行列・接続行列・隣接リストの 4 通りで表し、

 7    python graph_repr.py
 8"""
 9
10import networkx as nx
11import numpy as np
12import numpy.typing as npt
13
14# 頂点を 0, 1, ..., n-1 の整数で表す。
15Edge = tuple[int, int]
16AdjList = dict[int, list[int]]
17IntMatrix = npt.NDArray[np.int64]

20def to_adjacency_matrix(n: int, edges: list[Edge]) -> IntMatrix:
21    """辺リストから隣接行列を作る。A[u, v] は u と v を結ぶ辺の本数。"""
22    a = np.zeros((n, n), dtype=np.int64)
23    for u, v in edges:
24        a[u, v] += 1
25        a[v, u] += 1
26    return a
27
28
29def to_incidence_matrix(n: int, edges: list[Edge]) -> IntMatrix:
30    """辺リストから接続行列を作る。B[v, j] は頂点 v が辺 j の端点なら 1。"""
31    b = np.zeros((n, len(edges)), dtype=np.int64)
32    for j, (u, v) in enumerate(edges):
33        b[u, j] = 1
34        b[v, j] = 1
35    return b

39    """辺リストから隣接リストを作る。"""
40    adj: AdjList = {v: [] for v in range(n)}
41    for u, v in edges:
42        adj[u].append(v)
43        adj[v].append(u)
44    return adj

  1"""第 4 章 道と連結性 / 第 5 章 さまざまなグラフ.
  2
  3連結成分、距離と直径、二部グラフの判定、切断点と橋を求める。
  4最後に NetworkX の結果と照合する。
  5
  6実行方法:
  7    python connectivity.py
  8"""
  9
 10from collections import deque
 11from typing import cast
 12
 13import networkx as nx
 14
 15AdjList = dict[int, list[int]]
 16
 17
 18def build(n: int, edges: list[tuple[int, int]]) -> AdjList:
 19    """辺リストから無向グラフの隣接リストを作る。"""
 20    adj: AdjList = {v: [] for v in range(n)}
 21    for u, v in edges:
 22        adj[u].append(v)
 23        adj[v].append(u)
 24    return adj
 25
 26
 27def bfs_distances(adj: AdjList, source: int) -> dict[int, int]:
 28    """source から到達できる各頂点までの距離（辺の本数）を幅優先探索で求める。"""
 29    dist = {source: 0}
 30    queue = deque([source])
 31    while queue:
 32        u = queue.popleft()
 33        for v in adj[u]:
 34            if v not in dist:
 35                dist[v] = dist[u] + 1
 36                queue.append(v)
 37    return dist
 38
 39
 40def connected_components(adj: AdjList) -> list[set[int]]:
 41    """連結成分を頂点集合のリストとして返す。"""
 42    seen: set[int] = set()
 43    components: list[set[int]] = []
 44    for v in adj:
 45        if v not in seen:
 46            component = set(bfs_distances(adj, v))
 47            seen |= component
 48            components.append(component)
 49    return components
 50
 51
 52def diameter(adj: AdjList) -> int:
 53    """連結グラフの直径（2 頂点間の距離の最大値）を求める。"""
 54    return max(max(bfs_distances(adj, v).values()) for v in adj)
 55
 56
 57def bipartition(adj: AdjList) -> tuple[set[int], set[int]] | None:
 58    """二部グラフなら 2 つの頂点集合を、そうでなければ None を返す。
 59
 60    各連結成分を幅優先探索し、隣り合う頂点に異なる色 (0 / 1) を塗る。
 61    同じ色の頂点同士が隣接していたら、奇数長の閉路があるので二部グラフではない。
 62    """
 63    color: dict[int, int] = {}
 64    for start in adj:
 65        if start in color:
 66            continue
 67        color[start] = 0
 68        queue = deque([start])
 69        while queue:
 70            u = queue.popleft()
 71            for v in adj[u]:
 72                if v not in color:
 73                    color[v] = 1 - color[u]
 74                    queue.append(v)
 75                elif color[v] == color[u]:
 76                    return None
 77    left = {v for v, c in color.items() if c == 0}
 78    return left, set(adj) - left
 79
 80
 81def articulation_points_and_bridges(adj: AdjList) -> tuple[set[int], set[tuple[int, int]]]:
 82    """深さ優先探索の訪問順 order と lowlink 値 low から切断点と橋を求める。
 83
 84    low[v] は、v から深さ優先探索木の辺を 0 本以上下り、最後に後退辺を高々 1 本たどって
 85    到達できる頂点の order の最小値である。
 86    """
 87    order: dict[int, int] = {}
 88    low: dict[int, int] = {}
 89    points: set[int] = set()
 90    bridges: set[tuple[int, int]] = set()
 91
 92    def dfs(u: int, parent: int | None) -> None:
 93        order[u] = low[u] = len(order)
 94        children = 0
 95        for v in adj[u]:
 96            if v not in order:
 97                children += 1
 98                dfs(v, u)
 99                low[u] = min(low[u], low[v])
100                if parent is not None and low[v] >= order[u]:
101                    points.add(u)
102                if low[v] > order[u]:
103                    bridges.add((min(u, v), max(u, v)))
104            elif v != parent:
105                low[u] = min(low[u], order[v])
106        if parent is None and children >= 2:
107            points.add(u)
108
109    for v in adj:
110        if v not in order:
111            dfs(v, None)
112    return points, bridges
113
114
115def main() -> None:
116    # 2 つの三角形 {0, 1, 2} と {3, 4, 5} を辺 (2, 3) でつなぎ、頂点 5 に 6 をぶら下げたグラフ
117    # と、それとは別の連結成分 {7, 8}
118    n = 9
119    edges = [(0, 1), (1, 2), (2, 0), (2, 3), (3, 4), (4, 5), (5, 3), (5, 6), (7, 8)]
120    adj = build(n, edges)
121
122    print("連結成分:", connected_components(adj))
123    print("頂点 0 からの距離:", bfs_distances(adj, 0))
124    points, bridges = articulation_points_and_bridges(adj)
125    print("切断点:", sorted(points))
126    print("橋:", sorted(bridges))
127
128    # 6 頂点の閉路 C6 は二部グラフ、5 頂点の閉路 C5 は二部グラフでない
129    c6 = build(6, [(i, (i + 1) % 6) for i in range(6)])
130    c5 = build(5, [(i, (i + 1) % 5) for i in range(5)])
131    print("C6 の二部分割:", bipartition(c6))
132    print("C5 の二部分割:", bipartition(c5))
133    print("C6 の直径:", diameter(c6))
134
135    # NetworkX との照合
136    g: nx.Graph[int] = nx.Graph()
137    g.add_nodes_from(range(n))
138    g.add_edges_from(edges)
139    print("連結成分が一致:",
140          sorted(map(sorted, nx.connected_components(g))) == sorted(map(sorted, connected_components(adj))))
141    print("切断点が一致:", set(nx.articulation_points(g)) == points)
142    # 型スタブでは nx.bridges の戻り値の型が辺のタプルになっていないため cast で補う
143    nx_bridges = cast(list[tuple[int, int]], list(nx.bridges(g)))
144    print("橋が一致:", {(min(u, v), max(u, v)) for u, v in nx_bridges} == bridges)
145    print("C5 は二部グラフか (NetworkX):", nx.is_bipartite(nx.cycle_graph(5)))
146
147
148if __name__ == "__main__":
149    main()

spanning_tree.py

  1"""第 6 章 木と全域木.
  2
  3Union-Find、クラスカル法、プリム法で最小全域木を求める。
  4また、プリューファー符号で木と数列を相互に変換し、ケイリーの公式を確かめる。
  5
  6実行方法:
  7    python spanning_tree.py
  8"""
  9
 10import heapq
 11import itertools
 12from collections import Counter
 13
 14import networkx as nx
 15
 16# 重み付き辺 (重み, 端点 u, 端点 v)
 17WeightedEdge = tuple[float, int, int]
 18
 19
 20class UnionFind:
 21    """互いに素な集合を管理するデータ構造（経路圧縮とランクによる併合付き）。"""
 22
 23    def __init__(self, n: int) -> None:
 24        self.parent = list(range(n))
 25        self.rank = [0] * n
 26
 27    def find(self, x: int) -> int:
 28        """x が属する集合の代表元を返す。"""
 29        while self.parent[x] != x:
 30            self.parent[x] = self.parent[self.parent[x]]  # 経路圧縮（経路半分化）
 31            x = self.parent[x]
 32        return x
 33
 34    def union(self, x: int, y: int) -> bool:
 35        """x と y の集合を併合する。既に同じ集合なら False を返す。"""
 36        rx, ry = self.find(x), self.find(y)
 37        if rx == ry:
 38            return False
 39        if self.rank[rx] < self.rank[ry]:
 40            rx, ry = ry, rx
 41        self.parent[ry] = rx
 42        if self.rank[rx] == self.rank[ry]:
 43            self.rank[rx] += 1
 44        return True
 45
 46
 47def kruskal(n: int, edges: list[WeightedEdge]) -> list[WeightedEdge]:
 48    """クラスカル法: 重みの小さい辺から順に、閉路を作らないものを採用する。"""
 49    uf = UnionFind(n)
 50    tree: list[WeightedEdge] = []
 51    for w, u, v in sorted(edges):
 52        if uf.union(u, v):
 53            tree.append((w, u, v))
 54            if len(tree) == n - 1:
 55                break
 56    return tree
 57
 58
 59def prim(n: int, edges: list[WeightedEdge], root: int = 0) -> list[WeightedEdge]:
 60    """プリム法: 木に含まれる頂点集合から外へ出る最小の辺を繰り返し採用する。"""
 61    adj: dict[int, list[tuple[float, int]]] = {v: [] for v in range(n)}
 62    for w, u, v in edges:
 63        adj[u].append((w, v))
 64        adj[v].append((w, u))
 65    in_tree = {root}
 66    heap = [(w, root, v) for w, v in adj[root]]
 67    heapq.heapify(heap)
 68    tree: list[WeightedEdge] = []
 69    while heap and len(in_tree) < n:
 70        w, u, v = heapq.heappop(heap)
 71        if v in in_tree:
 72            continue
 73        in_tree.add(v)
 74        tree.append((w, u, v))
 75        for w2, x in adj[v]:
 76            if x not in in_tree:
 77                heapq.heappush(heap, (w2, v, x))
 78    return tree
 79

 81def prufer_encode(n: int, tree_edges: list[tuple[int, int]]) -> list[int]:
 82    """頂点 0..n-1 の木をプリューファー符号（長さ n-2 の数列）に変換する。"""
 83    adj: dict[int, set[int]] = {v: set() for v in range(n)}
 84    for u, v in tree_edges:
 85        adj[u].add(v)
 86        adj[v].add(u)
 87    code: list[int] = []
 88    for _ in range(n - 2):
 89        leaf = min(v for v in adj if len(adj[v]) == 1)  # 番号最小の葉
 90        neighbor = adj[leaf].pop()
 91        code.append(neighbor)
 92        adj[neighbor].discard(leaf)
 93        del adj[leaf]
 94    return code
 95
 96
 97def prufer_decode(code: list[int]) -> list[tuple[int, int]]:
 98    """プリューファー符号から木の辺リストを復元する。"""
 99    n = len(code) + 2
100    remaining = Counter(code)  # 残りの符号に各頂点が何回現れるか
101    edges: list[tuple[int, int]] = []
102    used: set[int] = set()
103    for x in code:
104        leaf = min(v for v in range(n) if v not in used and remaining[v] == 0)
105        edges.append((leaf, x))
106        used.add(leaf)
107        remaining[x] -= 1
108    u, v = (w for w in range(n) if w not in used)
109    edges.append((u, v))
110    return edges
111
112
113def main() -> None:
114    n = 6
115    edges: list[WeightedEdge] = [
116        (7, 0, 1), (5, 0, 3), (8, 1, 2), (9, 1, 3), (7, 1, 4),
117        (5, 2, 4), (15, 3, 4), (6, 3, 5), (8, 4, 5), (11, 2, 5),
118    ]
119    tk = kruskal(n, edges)
120    tp = prim(n, edges)
121    print("クラスカル法:", tk, "重み合計 =", sum(w for w, _, _ in tk))
122    print("プリム法    :", tp, "重み合計 =", sum(w for w, _, _ in tp))
123
124    g: nx.Graph[int] = nx.Graph()
125    for w, u, v in edges:
126        g.add_edge(u, v, weight=w)
127    mst = nx.minimum_spanning_tree(g)
128    print("NetworkX    : 重み合計 =", mst.size(weight="weight"))
129
130    # プリューファー符号の往復変換
131    tree = [(0, 3), (1, 3), (2, 3), (3, 4), (4, 5)]
132    code = prufer_encode(6, tree)
133    print("プリューファー符号:", code)
134    print("復元した木:", sorted(tuple(sorted(e)) for e in prufer_decode(code)))
135
136    # ケイリーの公式: 頂点数 n のラベル付き木は n^(n-2) 個
137    # 長さ n-2 の数列すべてを復元し、異なる木が n^(n-2) 個得られることを確かめる
138    for m in range(2, 7):
139        trees = {frozenset(frozenset(e) for e in prufer_decode(list(c)))
140                 for c in itertools.product(range(m), repeat=m - 2)}
141        print(f"n = {m}: 木の個数 = {len(trees)}, n^(n-2) = {m ** (m - 2)}")
142
143
144if __name__ == "__main__":
145    main()

  9
 10import networkx as nx
 11
 12Edge = tuple[int, int]
 13
 14
 15def odd_vertices(n: int, edges: list[Edge]) -> list[int]:
 16    """次数が奇数の頂点の一覧を返す。"""
 17    deg = [0] * n
 18    for u, v in edges:
 19        deg[u] += 1
 20        deg[v] += 1
 21    return [v for v in range(n) if deg[v] % 2 == 1]
 22
 23
 24def euler_path(n: int, edges: list[Edge]) -> list[int] | None:
 25    """ヒールホルツァーのアルゴリズムでオイラー路（またはオイラー閉路）を求める。
 26
 27    多重辺を扱えるよう、辺を番号で区別する。オイラー路が存在しなければ None を返す。
 28    """
 29    odd = odd_vertices(n, edges)
 30    if len(odd) not in (0, 2):
 31        return None
 32    adj: dict[int, list[tuple[int, int]]] = {v: [] for v in range(n)}  # 頂点 -> [(隣接頂点, 辺番号)]
 33    for i, (u, v) in enumerate(edges):
 34        adj[u].append((v, i))
 35        adj[v].append((u, i))
 36    used = [False] * len(edges)
 37    start = odd[0] if odd else edges[0][0]
 38
 39    stack = [start]
 40    path: list[int] = []
 41    while stack:
 42        u = stack[-1]
 43        # u から出る未使用の辺を 1 本たどる。行き止まりになったら path に確定させる。
 44        while adj[u] and used[adj[u][-1][1]]:
 45            adj[u].pop()
 46        if adj[u]:
 47            v, i = adj[u].pop()
 48            used[i] = True
 49            stack.append(v)
 50        else:
 51            path.append(stack.pop())
 52    if len(path) != len(edges) + 1:  # 辺を持つ部分が連結でない
 53        return None
 54    return path[::-1]

 59    nodes = list(g.nodes)
 60    n = len(nodes)
 61    start = nodes[0]
 62    path = [start]
 63    visited = {start}
 64
 65    def extend() -> bool:
 66        if len(path) == n:
 67            return g.has_edge(path[-1], start)
 68        for v in g.neighbors(path[-1]):
 69            if v not in visited:
 70                path.append(v)
 71                visited.add(v)
 72                if extend():
 73                    return True
 74                path.pop()
 75                visited.remove(v)
 76        return False

 60def chromatic_polynomial(edges: frozenset[frozenset[int]], n: int) -> list[int]:
 61    """削除縮約公式 P(G) = P(G - e) - P(G / e) で彩色多項式の係数を求める。
 62
 63    戻り値の i 番目の要素が k^i の係数である。辺を持たない n 頂点のグラフは k^n。
 64    """
 65    if not edges:
 66        return [0] * n + [1]

 68    u, v = sorted(e)
 69    deleted = edges - {e}
 70    # 縮約: v を u に併合する。生じる自己ループは e 自身なので除かれ、多重辺は集合なので 1 本になる
 71    contracted = frozenset(frozenset(u if x == v else x for x in f) for f in deleted)
 72    p1 = chromatic_polynomial(deleted, n)
 73    p2 = chromatic_polynomial(contracted, n - 1)
 74    return [a - (p2[i] if i < len(p2) else 0) for i, a in enumerate(p1)]

 13def max_bipartite_matching(left: list[str], adj: dict[str, list[str]]) -> dict[str, str]:
 14    """増加路法で二部グラフの最大マッチングを求める。
 15
 16    adj[u] は左側の頂点 u に隣接する右側の頂点のリスト。
 17    戻り値は右側の頂点 -> 左側の頂点 の対応。
 18    """
 19    match_of: dict[str, str] = {}
 20
 21    def try_augment(u: str, visited: set[str]) -> bool:
 22        """u から始まる増加路を深さ優先探索で探し、見つかればマッチングを更新する。"""
 23        for v in adj[u]:
 24            if v in visited:
 25                continue
 26            visited.add(v)
 27            # v が未マッチ、または v の相手が別の右側頂点に付け替えられるなら増加路がある
 28            if v not in match_of or try_augment(match_of[v], visited):
 29                match_of[v] = u
 30                return True
 31        return False
 32
 33    for u in left:
 34        try_augment(u, set())
 35    return match_of

 38def min_vertex_cover(left: list[str], adj: dict[str, list[str]], match_of: dict[str, str]) -> set[str]:
 39    """ケーニッヒの定理の証明に沿って、最大マッチングから最小頂点被覆を作る。
 40
 41    左側の未マッチ頂点から交互路（非マッチ辺で右へ、マッチ辺で左へ）で到達できる頂点を Z とすると、
 42    (左側 - Z) ∪ (右側 ∩ Z) が最小頂点被覆になる。
 43    """
 44    matched_left = set(match_of.values())
 45    reached = {u for u in left if u not in matched_left}
 46    stack = list(reached)
 47    while stack:
 48        u = stack.pop()
 49        for v in adj[u]:
 50            if v not in reached:
 51                reached.add(v)
 52                w = match_of.get(v)
 53                if w is not None and w not in reached:
 54                    reached.add(w)
 55                    stack.append(w)
 56    right = {v for vs in adj.values() for v in vs}
 57    return (set(left) - reached) | (right & reached)

 60def gale_shapley(proposer_pref: dict[str, list[str]], acceptor_pref: dict[str, list[str]]) -> dict[str, str]:
 61    """ゲール・シャプレー法（受け入れ保留方式）で安定マッチングを求める。
 62
 63    戻り値は 受け入れ側 -> 提案側 の対応。
 64    """
 65    rank = {a: {p: i for i, p in enumerate(prefs)} for a, prefs in acceptor_pref.items()}
 66    next_index = {p: 0 for p in proposer_pref}  # 次に提案する相手の順位
 67    engaged: dict[str, str] = {}
 68    free = list(proposer_pref)
 69    while free:
 70        p = free.pop()
 71        a = proposer_pref[p][next_index[p]]
 72        next_index[p] += 1
 73        current = engaged.get(a)
 74        if current is None:
 75            engaged[a] = p
 76        elif rank[a][p] < rank[a][current]:
 77            engaged[a] = p
 78            free.append(current)  # 振られた側は再び提案する
 79        else:
 80            free.append(p)
 81    return engaged

 72    components: list[set[str]] = []
 73
 74    def visit(u: str) -> None:
 75        index[u] = low[u] = len(index)
 76        stack.append(u)
 77        on_stack.add(u)
 78        for v in adj[u]:
 79            if v not in index:
 80                visit(v)
 81                low[u] = min(low[u], low[v])
 82            elif v in on_stack:
 83                low[u] = min(low[u], index[v])
 84        if low[u] == index[u]:  # u は強連結成分の根

 87                w = stack.pop()
 88                on_stack.remove(w)
 89                component.add(w)
 90                if w == u:
 91                    break
 92            components.append(component)
 93

 29        d, u = heapq.heappop(heap)
 30        if d > dist[u]:  # 古いエントリは読み飛ばす
 31            continue
 32        for v, w in adj[u]:
 33            if d + w < dist[v]:
 34                dist[v] = d + w
 35                parent[v] = u
 36                heapq.heappush(heap, (dist[v], v))
 37    return dist, parent

 48def bellman_ford(adj: WeightedAdj, source: str) -> dict[str, float] | None:
 49    """ベルマン・フォード法。source から到達できる負閉路があれば None を返す。"""
 50    dist = {v: math.inf for v in adj}
 51    dist[source] = 0.0
 52    edges = [(u, v, w) for u in adj for v, w in adj[u]]
 53    for _ in range(len(adj) - 1):
 54        updated = False
 55        for u, v, w in edges:
 56            if dist[u] + w < dist[v]:
 57                dist[v] = dist[u] + w
 58                updated = True
 59        if not updated:
 60            break
 61    # n - 1 回の反復後もまだ短くできる辺があれば負閉路が存在する
 62    if any(dist[u] + w < dist[v] for u, v, w in edges):
 63        return None
 64    return dist

 67def floyd_warshall(adj: WeightedAdj) -> dict[str, dict[str, float]]:
 68    """ワーシャル・フロイド法で全点対の最短距離を求める。"""
 69    nodes = list(adj)
 70    dist = {u: {v: (0.0 if u == v else math.inf) for v in nodes} for u in nodes}
 71    for u in adj:
 72        for v, w in adj[u]:
 73            dist[u][v] = min(dist[u][v], float(w))
 74    for k in nodes:  # 経由してよい頂点を 1 つずつ増やす
 75        for i in nodes:
 76            for j in nodes:
 77                if dist[i][k] + dist[k][j] < dist[i][j]:
 78                    dist[i][j] = dist[i][k] + dist[k][j]
 79    return dist

 85def astar_grid(grid: list[str], start: Cell, goal: Cell, use_heuristic: bool = True) -> tuple[int, int] | None:
 86    """格子上の A* 探索。'#' は壁。マンハッタン距離をヒューリスティックに用いる。
 87
 88    use_heuristic が False ならヒューリスティックを常に 0 とし、ダイクストラ法と同じ動作になる。
 89    戻り値は (最短距離, 展開した頂点数)。到達できなければ None。
 90    """
 91    def h(c: Cell) -> int:
 92        return abs(c[0] - goal[0]) + abs(c[1] - goal[1]) if use_heuristic else 0
 93
 94    g_cost = {start: 0}
 95    heap = [(h(start), start)]
 96    closed: set[Cell] = set()
 97    while heap:
 98        _, c = heapq.heappop(heap)
 99        if c in closed:
100            continue
101        closed.add(c)
102        if c == goal:
103            return g_cost[c], len(closed)
104        r, col = c
105        for nr, nc in ((r + 1, col), (r - 1, col), (r, col + 1), (r, col - 1)):
106            if 0 <= nr < len(grid) and 0 <= nc < len(grid[0]) and grid[nr][nc] != "#":
107                ng = g_cost[c] + 1
108                if ng < g_cost.get((nr, nc), math.inf):
109                    g_cost[(nr, nc)] = ng
110                    heapq.heappush(heap, (ng + h((nr, nc)), (nr, nc)))
111    return None

17def edmonds_karp(capacity: Capacity, s: str, t: str) -> tuple[int, Capacity]:
18    """最大フローの値と、各辺の流量を返す。
19
20    簡単のため、u -> v と v -> u の両方の辺を持つ頂点の組は無いものとする。
21    """
22    # 残余容量: 順方向は容量、逆方向は 0 から始める
23    residual: Capacity = {u: {} for u in capacity}
24    for u in capacity:
25        for v, c in capacity[u].items():
26            residual.setdefault(v, {})
27            residual[u][v] = residual[u].get(v, 0) + c
28            residual[v].setdefault(u, 0)
29
30    total = 0
31    while True:
32        # 幅優先探索で、残余容量が正の辺だけを通る s-t 路を探す
33        parent: dict[str, str] = {s: s}
34        queue = deque([s])
35        while queue and t not in parent:
36            u = queue.popleft()
37            for v, r in residual[u].items():
38                if r > 0 and v not in parent:
39                    parent[v] = u
40                    queue.append(v)
41        if t not in parent:
42            break
43        # 増加路上の残余容量の最小値だけ流す
44        path = [t]
45        while path[-1] != s:
46            path.append(parent[path[-1]])
47        path.reverse()
48        delta = min(residual[u][v] for u, v in zip(path, path[1:]))
49        for u, v in zip(path, path[1:]):
50            residual[u][v] -= delta
51            residual[v][u] += delta
52        total += delta
53
54    flow = {u: {v: c - residual[u][v] for v, c in capacity[u].items()} for u in capacity}
55    return total, flow

58def min_cut(capacity: Capacity, flow: Capacity, s: str) -> tuple[set[str], list[tuple[str, str]]]:
59    """最大フローの残余グラフで s から到達できる頂点集合 S と、S から出る辺（カット）を返す。"""
60    reach = {s}
61    stack = [s]
62    while stack:
63        u = stack.pop()
64        candidates = [(v, capacity[u][v] - flow[u][v]) for v in capacity.get(u, {})]
65        candidates += [(w, flow[w][u]) for w in capacity if u in capacity[w]]  # 逆向きに戻せる流量
66        for v, r in candidates:
67            if r > 0 and v not in reach:
68                reach.add(v)
69                stack.append(v)
70    cut = [(u, v) for u in reach for v in capacity.get(u, {}) if v not in reach]
71    return reach, cut

32def count_spanning_trees(a: FloatMatrix) -> int:
33    """行列木定理: ラプラシアン行列から 1 行 1 列を除いた行列式が全域木の個数に等しい。"""
34    reduced = laplacian(a)[1:, 1:]

38def fiedler_partition(a: FloatMatrix) -> tuple[float, list[int], list[int]]:
39    """2 番目に小さい固有値（代数的連結度）と、フィードラーベクトルの符号による 2 分割を返す。"""
40    eigenvalues, eigenvectors = np.linalg.eigh(laplacian(a))  # 固有値は昇順に並ぶ
41    fiedler = eigenvectors[:, 1]
42    plus = [i for i in range(len(fiedler)) if fiedler[i] >= 0]
43    minus = [i for i in range(len(fiedler)) if fiedler[i] < 0]
44    return float(eigenvalues[1]), plus, minus

47def pagerank(links: dict[int, list[int]], damping: float = 0.85, tol: float = 1e-10) -> FloatVector:
48    """べき乗法で PageRank を求める。リンクを持たないページからは全ページへ均等に移るものとする。"""
49    n = len(links)
50    p = np.zeros((n, n))  # p[j, i] は i から j へ移る確率
51    for i, outs in links.items():
52        if outs:
53            for j in outs:
54                p[j, i] = 1.0 / len(outs)
55        else:
56            p[:, i] = 1.0 / n
57    x = np.full(n, 1.0 / n)
58    while True:
59        x_new: FloatVector = damping * p @ x + (1.0 - damping) / n
60        if np.abs(x_new - x).sum() < tol:
61            return x_new
62        x = x_new
